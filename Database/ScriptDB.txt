CREATE DATABASE LibraryManagement;
GO

USE LibraryManagement;
GO

-- ==========================================
-- 1. Bảng Category (Thể loại)
-- ==========================================
CREATE TABLE Category (
    CategoryID INT IDENTITY(1,1) PRIMARY KEY,
    CategoryName NVARCHAR(100) NOT NULL
);
GO

-- ==========================================
-- 2. Bảng Author (Tác giả)
-- ==========================================
CREATE TABLE Author (
    AuthorID INT IDENTITY(1,1) PRIMARY KEY,
    AuthorName NVARCHAR(100) NOT NULL
);
GO

-- ==========================================
-- 3. Bảng Book (Sách)
-- ==========================================
CREATE TABLE Book (
    BookID INT IDENTITY(1,1) PRIMARY KEY,
    Title NVARCHAR(200) NOT NULL,
    CategoryID INT NOT NULL,
    AuthorID INT NOT NULL,
    Price DECIMAL(18,2) NOT NULL
        CHECK (Price >= 0),
    StockQuantity INT NOT NULL
        CHECK (StockQuantity >= 0),
    ImageUrl VARCHAR(500),
    Description NVARCHAR(MAX),
    IsActive BIT DEFAULT 1,

    CONSTRAINT FK_Book_Category
        FOREIGN KEY (CategoryID)
        REFERENCES Category(CategoryID),

    CONSTRAINT FK_Book_Author
        FOREIGN KEY (AuthorID)
        REFERENCES Author(AuthorID)
);
GO

-- ==========================================
-- 4. Bảng Book_Author (Liên kết sách - Tác giả)
-- ==========================================
CREATE TABLE Book_Author (
    BookID INT NOT NULL,
    AuthorID INT NOT NULL,

    CONSTRAINT PK_Book_Author
        PRIMARY KEY (BookID, AuthorID),

    CONSTRAINT FK_BookAuthor_Book
        FOREIGN KEY (BookID)
        REFERENCES Book(BookID)
        ON DELETE CASCADE,

    CONSTRAINT FK_BookAuthor_Author
        FOREIGN KEY (AuthorID)
        REFERENCES Author(AuthorID)
        ON DELETE CASCADE
);
GO

-- ==========================================
-- 5. Bảng Member (Thành viên)
-- ==========================================
CREATE TABLE Member (
    MemberID INT IDENTITY(1,1) PRIMARY KEY,
    FullName NVARCHAR(100) NOT NULL,
    Email VARCHAR(100) UNIQUE,
    Phone VARCHAR(20),
    Address NVARCHAR(255),
    CardExpiryDate DATE,
    IsLocked BIT DEFAULT 0
);
GO

-- ==========================================
-- 6. Bảng Loan (Phiếu mượn)
-- ==========================================
CREATE TABLE Loan (
    LoanID INT IDENTITY(1,1) PRIMARY KEY,
    MemberID INT NOT NULL,
    LoanDate DATE NOT NULL DEFAULT GETDATE(),
    DueDate DATE NOT NULL,
    ReturnDate DATE,
    TotalFine DECIMAL(18,2) DEFAULT 0
        CHECK (TotalFine >= 0),

    CONSTRAINT FK_Loan_Member
        FOREIGN KEY (MemberID)
        REFERENCES Member(MemberID),

    CONSTRAINT CHK_DueDate
        CHECK (DueDate >= LoanDate),

    CONSTRAINT CHK_ReturnDate
        CHECK (
            ReturnDate IS NULL
            OR ReturnDate >= LoanDate
        )
);
GO

-- ==========================================
-- 7. Bảng LoanDetail (Chi tiết phiếu mượn)
-- ==========================================
CREATE TABLE LoanDetail (
    LoanDetailID INT IDENTITY(1,1) PRIMARY KEY,
    LoanID INT NOT NULL,
    BookID INT NOT NULL,
    Quantity INT NOT NULL CHECK (Quantity > 0),
    FineAmount DECIMAL(18,2) DEFAULT 0
        CHECK (FineAmount >= 0),

    CONSTRAINT FK_LoanDetail_Loan
        FOREIGN KEY (LoanID)
        REFERENCES Loan(LoanID)
        ON DELETE CASCADE,

    CONSTRAINT FK_LoanDetail_Book
        FOREIGN KEY (BookID)
        REFERENCES Book(BookID)
);
GO

-- ==========================================
-- 12. Bảng Account (Phân quyền)
-- Role: Admin, Librarian, Reader
-- ==========================================
CREATE TABLE Account (
    AccountID INT IDENTITY(1,1) PRIMARY KEY,
    Username VARCHAR(50) UNIQUE NOT NULL,
    Password VARCHAR(255) NOT NULL,
    Role NVARCHAR(50) NOT NULL
);
GO

-- ==========================================
-- 13. Bảng Reservation (Đặt trước sách)
-- ==========================================
CREATE TABLE Reservation (
    ReserveID INT IDENTITY(1,1) PRIMARY KEY,
    MemberID INT NOT NULL,
    BookID INT NOT NULL,
    ReserveDate DATETIME DEFAULT GETDATE(),
    Status NVARCHAR(50) DEFAULT N'Chờ xử lý',

    CONSTRAINT FK_Reservation_Member
        FOREIGN KEY (MemberID)
        REFERENCES Member(MemberID),

    CONSTRAINT FK_Reservation_Book
        FOREIGN KEY (BookID)
        REFERENCES Book(BookID)
);
GO
